Crie um algoritmo em linguagem natural para as três tarefas a seguir.  Organize o passo-a-passo de forma clara e lógica.
# Enviar uma mensagem pelo WhatsApp.
// Algoritmo: Enviar uma mensagem pelo WhatsApp
#include <stdio.h>

int main() {
    // Passo 1: Início
    printf("Iniciando processo de envio de mensagem pelo WhatsApp...\n");

    // Passo 2: Abrir o aplicativo
    printf("Abrindo aplicativo WhatsApp...\n");

    // Passo 3: Selecionar contato ou grupo
    printf("Procurando contato ou grupo desejado...\n");
    printf("Selecionando conversa...\n");

    // Passo 4: Digitar a mensagem
    printf("Digitando mensagem...\n");

    // Passo 5: Enviar a mensagem
    printf("Enviando mensagem...\n");

    // Passo 6: Confirmar envio
    printf("Mensagem enviada com sucesso!\n");

    // Passo 7: Fim
    printf("Processo concluído.\n");

    return 0;
}
